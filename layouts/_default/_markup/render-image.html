{{- $src := .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- $alignValues := slice "left" "right" "center" -}} {{- $align := "" -}} {{- $caption := $title -}} {{- if in $alignValues $title -}} {{- $align = $title -}} {{- $caption = "" -}} {{- end -}} {{- $u := urls.Parse $src -}} {{- $fragment := lower $u.Fragment -}} {{- $themeClass := "" -}} {{- if or (eq $fragment "light") (eq $fragment "gh-light-mode-only") -}} {{- $themeClass = "img-light-only" -}} {{- else if or (eq $fragment "dark") (eq $fragment "gh-dark-mode-only") -}} {{- $themeClass = "img-dark-only" -}} {{- end -}} {{- $width := "" -}} {{- $height := "" -}} {{- if not $u.IsAbs -}} {{- $cleanPath := strings.TrimPrefix "/" $u.Path -}} {{- $pageRes := or (.Page.Resources.GetMatch $cleanPath) (.Page.Resources.GetMatch $u.Path) -}} {{- if $pageRes -}} {{- $width = $pageRes.Width -}} {{- $height = $pageRes.Height -}} {{- else -}} {{- $staticPath := printf "static/%s" $cleanPath -}} {{- if fileExists $staticPath -}} {{- $cfg := imageConfig $staticPath -}} {{- $width = $cfg.Width -}} {{- $height = $cfg.Height -}} {{- else if .Page.File -}} {{- $contentPath := path.Join "content" .Page.File.Dir $cleanPath -}} {{- if fileExists $contentPath -}} {{- $cfg := imageConfig $contentPath -}} {{- $width = $cfg.Width -}} {{- $height = $cfg.Height -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $dimAttr := "" -}} {{- if and $width $height -}} {{- $dimAttr = printf ` width="%v" height="%v"` $width $height -}} {{- end -}} {{- $classAttr := "" -}} {{- if $themeClass -}} {{- $classAttr = printf ` class="%s"` $themeClass -}} {{- end -}} {{- if $caption -}} {{- else if $align -}}
{{- else -}} {{- end -}}
