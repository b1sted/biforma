{{- $src := .src -}} {{- $page := .page -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- $clean := strings.TrimPrefix "/" $src -}} {{- $u := urls.Parse $src -}} {{- if not $u.IsAbs -}} {{- with $page.Resources.GetMatch $clean -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- else -}} {{- $staticPath := printf "static/%s" $clean -}} {{- if fileExists $staticPath -}} {{- $cfg := imageConfig $staticPath -}} {{- $width = $cfg.Width -}} {{- $height = $cfg.Height -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "width" $width "height" $height) -}}
